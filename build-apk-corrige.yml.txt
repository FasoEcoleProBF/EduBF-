name: Build EduBF APK
on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Recuperer le code
        uses: actions/checkout@v4

      - name: Installer Node
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Installer Java 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Preparer EduBF
        run: |
          mkdir -p www
          cp index.html www/index.html
          cp manifest.json www/manifest.json || true
          npm init -y
          npm install @capacitor/core @capacitor/cli @capacitor/android

      - name: Creer le projet Android
        run: |
          npx cap init EduBF com.fasoecolepro.edubf --web-dir=www
          npx cap add android
          npx cap sync

      - name: Construire l APK
        run: |
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Donner l APK
        uses: actions/upload-artifact@v4
        with:
          name: EduBF-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
